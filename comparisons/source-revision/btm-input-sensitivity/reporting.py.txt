from pathlib import Path
from hashlib import sha256
import gzip,json,shutil,time
import numpy as np
import pandas as pd
from src.empirical_isopach.models.mass_mixture.study import audited_data
from src.empirical_isopach.models.mass_mixture.posterior import volume_summary
from src.empirical_isopach.models.mass_mixture.source_checks import source_checks
root=Path('results/source-btm-input-20261107')
out=Path('comparisons/source-revision/btm-input-sensitivity')
logs=Path('results/source-dev-20261106')
while not (root/'report.json').exists():time.sleep(2)
out.mkdir(parents=True,exist_ok=True)
r=json.loads((root/'report.json').read_text());fit=r['fit']
a,ma=audited_data('data/eruptions/B-Tm_tephra.csv')
expected=a.subset(a.row_ids!=84);expected.censored[expected.row_ids==82]=True
used=pd.read_csv(root/'used-observations.csv')
np.testing.assert_array_equal(used.OriginalCSVRow,expected.row_ids)
np.testing.assert_allclose(np.log(used.Thickness),expected.log_value,rtol=0,atol=1e-12)
np.testing.assert_array_equal(used.Kind.eq('upper_limit'),expected.censored)
original_table=pd.read_csv('data/eruptions/B-Tm_tephra.csv').iloc[expected.row_ids-2]
np.testing.assert_allclose(used[['Lon','Lat','Thickness']],original_table[['Lon','Lat','Thickness']],rtol=0,atol=1e-12)
assert len(used)==83 and (used.Kind=='upper_limit').sum()==1
assert r['runner_sha256']==sha256(Path('benchmarks/btm_input_review.py').read_bytes()).hexdigest()
p=root/'posteriors'/fit['posterior_file'];assert sha256(p.read_bytes()).hexdigest()==fit['posterior_sha256']
with np.load(p) as z:samples={k:z[k] for k in z.files}
assert volume_summary(samples)==fit['volume']
r['source_checks']=source_checks(samples)
basepath=Path('results/source-production-20261106/b-tm')
old=json.loads((basepath/'baseline.json').read_text());ov=old['volume']
with np.load(basepath/'posteriors/baseline'/old['posterior_file']) as z:oldvol=np.exp(z['log_volume'])
(out/'report.json').write_text(json.dumps(r,indent=2,allow_nan=False)+'\n')
shutil.copy2(root/'used-observations.csv',out/'used-observations.csv')
shutil.copy2(p,out/'posterior.npz')
vol=np.exp(samples['log_volume']);chain,draw=np.indices(vol.shape)
(out/'volume_draws.csv.gz').write_bytes(gzip.compress(pd.DataFrame({'chain':chain.ravel(),'draw':draw.ravel(),'volume_km3':vol.ravel()}).to_csv(index=False).encode(),mtime=0))
v=fit['volume'];d=fit['diagnostics']
(out/'README.md').write_text(f'''# B-Tm: two source-backed distal observation changes

[Published source and row crosswalk](../input-inventory/b-tm-workbook-review/README.md) · [Geometry comparison](../README.md)

Treating Suigetsu as the published **<0.1 cm upper bound**, and removing Kushu's unsupported exact 0.1 cm value from the quantitative likelihood, gives **{v['median']:,.0f} km³** with a **{v['lower95']:,.0f}–{v['upper95']:,.0f} km³ conditional 95% interval**. Numerical checks **{'pass' if d['passed'] else 'remain provisional'}**. The unchanged-input source-focused calculation gives **{ov['median']:,.0f} km³ ({ov['lower95']:,.0f}–{ov['upper95']:,.0f} km³)**.

| Input interpretation | Quantitative observations | Median bulk km³ | Conditional 95% interval km³ |
|---|---:|---:|---:|
| Preserved accepted input | 84 exact entries | {ov['median']:,.0f} | {ov['lower95']:,.0f}–{ov['upper95']:,.0f} |
| Two-site source sensitivity | 82 exact entries + 1 upper bound | {v['median']:,.0f} | {v['lower95']:,.0f}–{v['upper95']:,.0f} |

![The two source-focused posterior distributions](comparison.png)

(logs/'btm-input-verification.json').write_text(json.dumps({'row82_upper_bound_and_row84_omission_only':True,'all_retained_coordinates_values_errors_and_original_identities_unchanged':True,'retained_rows':83,'n_exact':82,'n_upper_bounds':1,'posterior_checksum_and_quantiles_verified':True,'numerical_passed':d['passed'],'source_checks':r['source_checks'],'original_input_sha256':ma['input_sha256'],'runner_sha256':r['runner_sha256']},indent=2)+'\n')
