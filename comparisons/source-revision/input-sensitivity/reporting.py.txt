from pathlib import Path
from hashlib import sha256
import json,gzip,shutil
import numpy as np
import pandas as pd
from src.empirical_isopach.models.mass_mixture.study import audited_data
from src.empirical_isopach.models.mass_mixture.posterior import volume_summary
root=Path('results/source-aso-input-20261107');out=Path('comparisons/source-revision/input-sensitivity');out.mkdir(parents=True,exist_ok=True)
r=json.loads((root/'report.json').read_text());fit=r['fit']
a,ma=audited_data('data/eruptions/Aso-4_tephra.csv');b,mb=audited_data(root/'aso-4-row55-lower-bound.csv')
for key in ('xy','log_value','log_sd','row_ids'):
 np.testing.assert_allclose(getattr(a,key),getattr(b,key),rtol=0,atol=1e-10)
assert np.array_equal(np.where(a.censored != b.censored)[0],np.where(a.row_ids==55)[0])
assert np.array_equal(np.where(a.lower_censored != b.lower_censored)[0],np.where(a.row_ids==55)[0])
assert r['runner_sha256']==sha256(Path('benchmarks/aso_input_review.py').read_bytes()).hexdigest()
p=root/'posteriors'/fit['posterior_file'];assert sha256(p.read_bytes()).hexdigest()==fit['posterior_sha256']
with np.load(p) as z:samples={k:z[k] for k in z.files}
assert volume_summary(samples)==fit['volume']
old=json.loads(Path('archive/reports/aso-4/2026-11-06-displaced-global/report.json').read_text())['fits']['baseline']
with np.load('archive/reports/aso-4/2026-11-06-displaced-global/posterior.npz') as z:oldvol=np.exp(z['log_volume'])
for name in ['report.json','aso-4-row55-lower-bound.csv']:shutil.copy2(root/name,out/name)
shutil.copy2(p,out/'posterior.npz')
vol=np.exp(samples['log_volume']);chain,draw=np.indices(vol.shape)
frame=pd.DataFrame({'chain':chain.ravel(),'draw':draw.ravel(),'volume_km3':vol.ravel()})
(out/'volume_draws.csv.gz').write_bytes(gzip.compress(frame.to_csv(index=False).encode(),mtime=0))
v=fit['volume'];d=fit['diagnostics'];ov=old['volume']

Numerical checks **{'pass' if d['passed'] else 'remain provisional'}**. This one observation interpretation does **not** explain away the high Aso-4 volume.
